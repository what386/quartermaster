cmake_minimum_required(VERSION 3.16)
project(QuartermasterUpdater LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "The native updater supports Windows only.")
endif()

add_executable(Quartermaster.Updater main.cpp process.cpp replace.cpp)
target_compile_features(Quartermaster.Updater PRIVATE cxx_std_17)
set_target_properties(Quartermaster.Updater PROPERTIES CXX_EXTENSIONS OFF CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set_property(TARGET Quartermaster.Updater PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(Quartermaster.Updater PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(Quartermaster.Updater PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(Quartermaster.Updater PRIVATE $<$<CONFIG:MinSizeRel>:-s>)
endif()
install(TARGETS Quartermaster.Updater RUNTIME DESTINATION winupdater)

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(updater-test-app tests/app.cpp)
    target_compile_features(updater-test-app PRIVATE cxx_std_17)
    if(MSVC)
        set_property(TARGET updater-test-app PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_test(NAME updater-integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_updater.py"
        "$<TARGET_FILE:Quartermaster.Updater>" "$<TARGET_FILE:updater-test-app>")
endif()
